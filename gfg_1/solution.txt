class Solution {
public:
    string encode(vector<string>& arr) {
        string res = "";
        for (string s : arr) {
            res += to_string(s.length()) + "/:" + s;
        }
        return res;
    }

    vector<string> decode(string &s) {
        vector<string> res;
        int i = 0;
        while (i < s.length()) {
            int j = i;
            while (s[j] != '/') {
                j++;
            }
            int len = stoi(s.substr(i, j - i));
            i = j + 2; // skip "/:"
            res.push_back(s.substr(i, len));
            i += len;
        }
        return res;
    }
};
